; ---- OUTDEC: print integer in EAX as decimal, followed by newline ----\nOUTDEC:\n\tPUSH EBX\n\tPUSH ECX\n\tPUSH EDX
\n\tPUSH ESI
\n\tOR   EAX, EAX
\n\tJGE  OUTDEC_POSITIVE
\n\tNEG  EAX
\n\tPUSH EAX
\n\tSUB  ESP, 4
\n\tMOV  byte [ESP], '-'
\n\tMOV  EAX, 4
\n\tMOV  EBX, 1
\n\tMOV  ECX, ESP
\n\tMOV  EDX, 1
\n\tINT  0x80
\n\tADD  ESP, 4
\n\tPOP  EAX
OUTDEC_POSITIVE:
\n\tXOR  ECX, ECX
\n\tMOV  EBX, 10
OUTDEC_DIGIT_LOOP:
\n\tXOR  EDX, EDX
\n\tDIV  EBX
\n\tADD  DL, 30h
\n\tPUSH EDX
\n\tINC  ECX
\n\tTEST EAX, EAX
\n\tJNZ  OUTDEC_DIGIT_LOOP
\n\tMOV  ESI, ECX
\n\tMOV  EBX, 1
\n\tMOV  EDX, 1
OUTDEC_PRINT_LOOP:
\n\tTEST ESI, ESI
\n\tJZ   OUTDEC_NEWLINE
\n\tMOV  EAX, 4
\n\tMOV  ECX, ESP
\n\tINT  0x80
\n\tADD  ESP, 4
\n\tDEC  ESI
\n\tJMP  OUTDEC_PRINT_LOOP
OUTDEC_NEWLINE:
\n\tSUB  ESP, 4
\n\tMOV  byte [ESP], 10
\n\tMOV  EAX, 4
\n\tMOV  ECX, ESP
\n\tINT  0x80
\n\tADD  ESP, 4
\n\tPOP  ESI
\n\tPOP  EDX
\n\tPOP  ECX
\n\tPOP  EBX
\n\tRET
